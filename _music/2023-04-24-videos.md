    A simple, elegant caption looks good between video rows, after each row, or doesn't have to be there at all.
</div>

<script>
document.addEventListener('DOMContentLoaded', function () {
  const v = document.querySelector('#light-chaser-wrap video');
  if (v) {
    v.autoplay = false;              // ensure no autoplay
    v.removeAttribute('autoplay');   // in case the include injects it
    v.volume = 0.5;                  // 50%
  }
});
</script>
